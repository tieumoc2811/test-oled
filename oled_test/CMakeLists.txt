cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico)
include(pico_sdk_import.cmake)
project(oled_test C CXX ASM)
pico_sdk_init()
add_executable(oled_test main.c)
if(DEFINED OLED_H)
  target_compile_definitions(oled_test PRIVATE OLED_H=${OLED_H})
endif()
target_link_libraries(oled_test pico_stdlib hardware_i2c)
pico_enable_stdio_usb(oled_test 1)
pico_enable_stdio_uart(oled_test 0)
pico_add_extra_outputs(oled_test)
